# Suderra OS — PRODUCTION config.txt (U-Boot chainload, ADR-0007)
# kernel=u-boot.bin: Pi firmware U-Boot'u yükler; U-Boot's compiled-in bootcmd
# (board/suderra/uboot/suderra.env) picks the RAUC slot and boots the signed FIT.
# The device tree the firmware loads is U-Boot's control FDT: the boot partition
# carries the keyed copies (firmware-dtb/ of the image build). DEV build board/suderra/aarch64-rpi4/config.txt (kernel=Image)
# kullanır. Yalnız prod genimage/post-image bu dosyayı config.txt olarak yerleştirir.
#
# Suderra OS — Raspberry Pi 4 / CM4 firmware konfigürasyonu
#
# Bu dosya Pi firmware (start4.elf) tarafından okunur ve kernel yüklemesi
# öncesi düşük seviye donanım kurulumu yapar. Boot zinciri:
#
#   ROM bootloader -> bootcode.bin (Pi 4 SPI EEPROM) -> start4.elf
#       -> config.txt -> kernel=u-boot.bin -> U-Boot (bootcmd) -> imzalı FIT
#
# Referans: https://www.raspberrypi.com/documentation/computers/config_txt.html

# ----------------------------------------------------------------------------
# Kernel & ARM
# ----------------------------------------------------------------------------

# 64-bit ARM mode (Cortex-A72 / AArch64)
arm_64bit=1

# Buildroot tarafından üretilen kernel image adı
kernel=u-boot.bin

# Kernel cmdline ayrı dosyadan okunur
# (Buildroot rpi-firmware paketi cmdline.txt'yi /boot içine kopyalar)

# ----------------------------------------------------------------------------
# Device Tree
# ----------------------------------------------------------------------------
#
# Suderra OS aynı SD imajını üç farklı boardda boot edebilir:
#   - Pi 4 Model B           -> bcm2711-rpi-4-b.dtb
#   - CM4 (custom carrier)   -> bcm2711-rpi-cm4.dtb
#   - CM4 Official IO Board  -> bcm2711-rpi-cm4-io.dtb
#
# Pi firmware otomatik olarak doğru DTB'yi seçer (board revision üzerinden).
# Manuel override gerekirse aşağıdaki satırı aç:
#
# device_tree=bcm2711-rpi-cm4-io.dtb

# DTB'nin lineer yüklenmesini sağla (overlay'ler için gerekli)
arm_boost=1

# ----------------------------------------------------------------------------
# Konsol & Debug
# ----------------------------------------------------------------------------

# GPIO14/15 carry the agent's fieldbus (the RS485 HAT of a Pi 4 / CM4 is
# /dev/ttyAMA0, board/suderra/common/rootfs-overlay/etc/systemd/system/
# suderra-agent.service.d/10-board-rpi4-cm4.conf), so the production image has no
# kernel console on it (the signed command line names console=tty1, EI-049) and
# the firmware's second stage does not write there either. U-Boot's own output
# still uses the console UART enable_uart=1 sets up; it is output only (stdin is the
# null device, bootdelay is -2), and a Modbus byte on the bus can neither stop the
# boot nor reach a prompt (ADR-048 DEC-18.3). Whether that output may stay is a
# hardware check (docs/hardware/): a HAT that must stay silent at power-up needs
# enable_uart=0 once U-Boot's console is shown to be optional on the board.
enable_uart=1
uart_2ndstage=0

# ----------------------------------------------------------------------------
# GPU / Display
# ----------------------------------------------------------------------------

# Headless edge cihazı — GPU minimum bellek
gpu_mem=16

# HDMI default kapalı (gerekirse cihaz konfigürasyonu ile açılır)
hdmi_blanking=2

# ----------------------------------------------------------------------------
# Watchdog (BCM2835 hardware)
# ----------------------------------------------------------------------------
#
# Sistem 15 saniyede yanıt vermezse hardware reset.
# systemd watchdog ile birlikte çalışır (systemd.service WatchdogSec=).

dtparam=watchdog=on

# ----------------------------------------------------------------------------
# Endüstriyel I/O (Edge Agent için)
# ----------------------------------------------------------------------------

# I2C — sensör bus (BME280, DS3231 RTC, vb.)
dtparam=i2c_arm=on
dtparam=i2c1=on

# SPI — high-speed sensör / ADC
dtparam=spi=on

# PWM — kontrol çıkışları (fan, valve actuator)
# dtoverlay=pwm-2chan

# 1-Wire (DS18B20 sıcaklık sensörü)
# dtoverlay=w1-gpio

# ----------------------------------------------------------------------------
# Endüstriyel network — kablolu eth tercih
# ----------------------------------------------------------------------------

# Bluetooth kapalı (UART0 serbest, saldırı yüzeyi az)
dtoverlay=disable-bt

# Wi-Fi kapalı (gerçek deployment'ta — testte aç)
# dtoverlay=disable-wifi

# ----------------------------------------------------------------------------
# Performans
# ----------------------------------------------------------------------------

# CPU normal frekansta (overclock yok — endüstriyel kararlılık)
# arm_freq=1500   # Pi 4 default 1500 MHz, override etme

# Cooling — CM4 endüstriyel kabin için fan kontrolü
# dtparam=cooling_fan=on
# dtparam=fan_temp0=60000   # 60°C'de fan başlasın

# ----------------------------------------------------------------------------
# initramfs (Faz 3'te dm-verity için aktive)
# ----------------------------------------------------------------------------

# initramfs rootfs.cpio.gz followkernel
# auto_initramfs=1

# ----------------------------------------------------------------------------
# RTC (ADR-048 DEC-7.8; EI-046) — NOT ENABLED BY THIS FILE
# ----------------------------------------------------------------------------

# The Compute Module 4 IO Board's RTC is a PCF85063AT (not a DS3231) at 0x51 on
# the VideoCore I2C bus, behind `dtparam=i2c_vc=on` (CM4 IO Board datasheet,
# section 2.13). In production U-Boot hands the kernel the device tree of the
# signed FIT, not the one the firmware builds from this file, so a dtparam or
# dtoverlay here never reaches the kernel (aarch64-revpi-connect4/board-boot.conf,
# EI-024, EI-029): the board's RTC works on a production rpi4-cm4 image only
# when the FIT's own tree has the bus enabled, which an rpi4 board-boot.conf
# (overlay merged by fdtoverlay, as the Connect 4's) would do and none exists
# yet. Until then the board's time floor (suderra-time-state) is the only clock
# it has before the first NTS synchronization. The dev image's config.txt
# enables the bus.

# ----------------------------------------------------------------------------
# Boot davranışı
# ----------------------------------------------------------------------------

# Boot sırasında splash gizli (sessiz boot — endüstriyel)
disable_splash=1

# Boot delay kapalı
boot_delay=0
